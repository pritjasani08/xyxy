// Manually store 2nd no in the memory location C051

	   LXI H,C050
	   MOV A,M
	   INX H
	   SUB M
	   INX H
	   MOV M,A
	   HLT
# ORG C050

	   MOV A,M
	   INX H
	   SUB M
